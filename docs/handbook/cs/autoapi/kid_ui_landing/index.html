kid_ui_landing

Landing page for kid_ui that starts multiple kid_ui.py instances via kid_ui_multi.py, shows all running instances (IP and port) with the number of connected users, and lets users either pick an instance to connect to or click „Auto-connect“ to connect to the best available instance.

Usage:
python3 kid_ui_landing.py [-n 5] [–host 0.0.0.0] [–port 5000]

[–instance-port 5001] [–stream-port 6001] [–world ball] [–log-dir DIR]

let language = "cs";
let latestData = null;

function applyLanguage() {
  document.documentElement.lang = language;
  document.title = TEXT[language].pageTitle;
  for (const el of document.querySelectorAll("[data-i18n]"))
    el.textContent = TEXT[language][el.dataset.i18n];
  document.getElementById("language").textContent = TEXT[language].switchLanguage;
  if (latestData) render(latestData);
}

function toggleLanguage() {
  language = language === "cs" ? "en" : "cs";
  applyLanguage();
}

function targetUrl(inst, resId) {
  const host = (location.hostname && location.hostname !== "0.0.0.0") ? location.hostname : inst.ip;
  const base = location.protocol + "//" + host + ":" + inst.port;
  return resId ? base + "/?res=" + encodeURIComponent(resId) : base;
}

function displayAddress(inst) {
  const addrs = (inst.addresses && inst.addresses.length) ? inst.addresses : [inst.ip];
  if (location.hostname && location.hostname !== "localhost" && location.hostname !== "127.0.0.1") {
    return location.hostname + ":" + inst.port;
  }
  return addrs.map(ip => ip + ":" + inst.port).join(" / ");
}

async function reserveAndGo(payload, fallbackInst) {
  try {
    const resp = await fetch("/api/reserve", {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify(payload || {})
    });
    if (resp.ok) {
      const data = await resp.json();
      if (data.instance) {
        window.location.href = targetUrl(data.instance, data.res);
        return;
      }
    }
  } catch (e) {}
  if (fallbackInst) {
    window.location.href = targetUrl(fallbackInst);
  }
}

function connectTo(inst) {
  reserveAndGo({id: inst.id}, inst);
}

function autoConnect() {
  reserveAndGo({}, latestData && latestData.best);
}
